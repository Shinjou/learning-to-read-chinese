class LookTab extends ConsumerWidget {
  final TeachWordControllers controllers;
  final TeachWordState state;
  final ScreenInfo screenInfo;

  const LookTab({
    super.key,
    required this.controllers,
    required this.state,
    required this.screenInfo,
  });

  String get word => state.word;
  bool get isBpmf => state.isBpmf;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    debugPrint('Building LookTab - word: ${state.word}, nextStepId: ${state.nextStepId}, isLearned: ${state.isLearned}');

    return TeachWordTabBarView(
      content: SingleChildScrollView(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Flexible(
              fit: FlexFit.loose,
              child: Center(
                child: ConstrainedBox(
                  constraints: BoxConstraints(
                    maxWidth: screenInfo.screenWidth * 0.90,
                  ),
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: <Widget>[
                      _buildLookHeader(context, ref),
                      SizedBox(height: screenInfo.fontSize * 0.3),
                      _buildLookImage(context, ref),
                    ],
                  ),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildLookHeader(BuildContext context, WidgetRef ref) {
    return LeftRightSwitch(
      iconsColor: lightGray,
      iconsSize: screenInfo.fontSize * 2.0,
      rightBorder: true,
      middleWidget: const TeachWordCardTitle(
        sectionName: '看一看',
        iconsColor: lightGray,
      ),
      isFirst: true,
      isLast: false,
      /*
      onRightClicked: state.isLearned || state.nextStepId == TeachWordConstants.steps['goToListen']
          ? () async {
              debugPrint('Navigate next clicked in LookTab');
              await _handleNext(ref);
              controllers.navigateToNextTab();
            }
          : null,
      */
      onRightClicked: () async { // Always allow Right Click in LookTab
        debugPrint('Navigate next clicked in LookTab');
        await _handleNext(ref);
        controllers.navigateToNextTab();
      },
    );
  }

  Widget _buildLookImage(BuildContext context, WidgetRef ref) {
    return Flexible(
      fit: FlexFit.loose,
      child: Stack(
        alignment: Alignment.center,
        children: [
          Container(
            decoration: const BoxDecoration(
              color: darkBrown,
            ),
            child: Center(
              child: Image(
                width: max(17.6 * screenInfo.fontSize, 300.0),
                image: state.isBpmf
                    ? AssetImage('lib/assets/img/bopomo/${state.word}.png')
                    : AssetImage('lib/assets/img/oldWords/${state.word}.webp'),
                errorBuilder: (context, error, stackTrace) {
                  debugPrint('Error loading image: $error');
                  return Center(
                    child: Text(
                      state.word,
                      textAlign: TextAlign.center,
                      style: TextStyle(
                        fontSize: screenInfo.fontSize * 8.0,
                        color: backgroundColor,
                        fontWeight: FontWeight.w100,
                        fontFamily: state.isBpmf ? "BpmfOnly" : "BpmfIansui",
                      ),
                    ),
                  );
                },
              ),
            ),
          ),
        ],
      ),
    );
  }

  Future<void> _handleNext(WidgetRef ref) async {
    debugPrint('LookTab _handleNext called - nextStepId: ${state.nextStepId}');
    
    // Increment step before playing sound
    ref.read(teachWordStateProvider.notifier).incrementNextStepId();
    final audioService = ref.read(audioServiceProvider);
    audioService.playWordSound(word, isBpmf);
  }
}

class UseTab extends ConsumerStatefulWidget {
  final TeachWordControllers controllers;
  final TeachWordState state;
  final ScreenInfo screenInfo;
  final Map<String, dynamic> wordsPhrase;

  const UseTab({
    super.key,
    required this.controllers,
    required this.state,
    required this.screenInfo,
    required this.wordsPhrase,
  });

  @override
  UseTabState createState() => UseTabState();
}

class UseTabState extends ConsumerState<UseTab> {
  late final ValueNotifier<String> _userMessage;
  late final ValueNotifier<bool> _showContinueButton;

  @override
  void initState() {
    super.initState();
    _initializeControllers();
    debugPrint('UseTab initialized');
  }

  void _initializeControllers() {
    _userMessage = ValueNotifier('');
    _showContinueButton = ValueNotifier(false);
  }

  @override
  void dispose() {
    debugPrint('UseTab disposing');
    _userMessage.dispose();
    _showContinueButton.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    debugPrint('Building UseTab - nextStepId: ${widget.state.nextStepId}, vocabIndex: ${widget.state.vocabIndex}');
    
    return TeachWordTabBarView(
      content: Stack(
        children: [
          SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                _buildContent(),
              ],
            ),
          ),
          _buildPageIndicator(),
        ],
      ),
    );
  }

  Widget _buildContent() {
    return ConstrainedBox(
      constraints: BoxConstraints(
        maxWidth: widget.screenInfo.screenWidth * 0.95,
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: <Widget>[
          _buildHeader(),
          SizedBox(height: widget.screenInfo.fontSize * 0.2),
          _buildVocabContent(),
        ],
      ),
    );
  }

  Widget _buildHeader() {    
    return LeftRightSwitch(
      iconsColor: lightGray,
      iconsSize: widget.screenInfo.fontSize * 2.0,
      rightBorder: _shouldShowRightBorder,
      middleWidget: const TeachWordCardTitle(
        sectionName: '用一用',
        iconsColor: lightGray,
      ),
      isFirst: false,
      isLast: _isLastVocabPage,
      onLeftClicked: _canNavigatePrevious ? () => _handlePrevious() : null,
      onRightClicked: _canNavigateNext ? () => _handleNext() : null,
    );
  }

  Widget _buildVocabContent() {
    if (widget.state.isBpmf) {
      return BopomofoVocabContent(
        word: widget.state.word,
        vocab: _getCurrentVocab(),
        sentence: _getCurrentSentence(),
      );
    } else {
      return Column(
        children: [
          WordVocabContent(
            vocab: _getCurrentVocab(),
            meaning: _getCurrentMeaning(),
            sentence: _getCurrentSentence(),
            vocab2: _getNextVocab(),
          ),
          _buildVocabImage(),
          ValueListenableBuilder<String>(
            valueListenable: _userMessage,
            builder: (context, message, child) {
              return _buildMessageAndContinueButton(message);
            },
          ),
        ],
      );
    }
  }

  Widget _buildVocabImage() {
    if (!widget.state.isBpmf && _imageExistsForCurrentVocab) {
      return SizedBox(
        height: widget.screenInfo.fontSize * 3.0,
        child: Image(
          height: widget.screenInfo.fontSize * 3.0,
          image: AssetImage('lib/assets/img/vocabulary/${_getCurrentVocab()}.webp'),
          errorBuilder: (context, error, stackTrace) {
            debugPrint('Error loading vocab image: $error');
            return const SizedBox.shrink();
          },          
        ),
      );
    }
    return SizedBox(height: widget.screenInfo.fontSize * 0.5);
  }

  Widget _buildMessageAndContinueButton(String message) {
    return Padding(
      padding: EdgeInsets.symmetric(vertical: widget.screenInfo.fontSize * 0.5),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Text(
            message,
            style: TextStyle(
              color: explanationColor,
              fontSize: widget.screenInfo.fontSize,
            ),
          ),
          ValueListenableBuilder<bool>(
            valueListenable: _showContinueButton,
            builder: (context, show, child) {
              return show ? _buildContinueButton() : const SizedBox.shrink();
            },
          ),
        ],
      ),
    );
  }

  Widget _buildContinueButton() {
    return TextButton(
      onPressed: _handleContinue,
      child: Text(
        '繼續',
        style: TextStyle(
          fontSize: widget.screenInfo.fontSize,
          color: Colors.transparent,
        ),
      ),
    );
  }

  Widget _buildPageIndicator() {
    return Positioned(
      right: widget.screenInfo.fontSize,
      bottom: widget.screenInfo.fontSize,
      child: Text(
        "${widget.state.vocabIndex + 1} / ${widget.state.vocabCount}",
        style: TextStyle(
          fontSize: widget.screenInfo.fontSize * 0.75,
          fontWeight: FontWeight.normal,
          color: backgroundColor,
        ),
      ),
    );
  }

  // Helper methods
  String _getCurrentVocab() => widget.wordsPhrase['vocab${widget.state.vocabIndex + 1}'] ?? '';
  String _getCurrentMeaning() => widget.wordsPhrase['meaning${widget.state.vocabIndex + 1}'] ?? '';
  String _getCurrentSentence() => widget.wordsPhrase['sentence${widget.state.vocabIndex + 1}'] ?? '';
  String _getNextVocab() {
    int nextIndex = (widget.state.vocabIndex + 1) % widget.state.vocabCount + 1;
    return widget.wordsPhrase['vocab$nextIndex'] ?? '';
  }

  bool get _imageExistsForCurrentVocab => 
    widget.state.vocabIndex == 0 ? widget.state.img1Exists : widget.state.img2Exists;

  bool get _isLastVocabPage => 
    widget.state.vocabIndex == widget.state.vocabCount - 1;

  bool get _shouldShowRightBorder =>
    widget.state.vocabIndex == 0 && 
    widget.state.nextStepId == TeachWordConstants.steps['goToUse2'];

  bool get _canNavigatePrevious =>
    widget.state.isLearned || 
    (widget.state.vocabIndex == 0 
      ? widget.state.nextStepId == TeachWordConstants.steps['goToUse2'] 
      : true);

  bool get _canNavigateNext =>
    !_isLastVocabPage && 
    (widget.state.nextStepId == TeachWordConstants.steps['goToUse${widget.state.vocabIndex + 2}'] || 
     widget.state.isLearned);

  // Event handlers
  Future<void> _handlePrevious() async {
    debugPrint('UseTab handlePrevious called - vocabIndex: ${widget.state.vocabIndex}');
    if (widget.state.vocabIndex == 0) {
      widget.controllers.navigateToPreviousTab();
    } else {
      await _speakPreviousVocab();
      ref.read(teachWordStateProvider.notifier).updateVocabIndex(
        widget.state.vocabIndex - 1
      );
    }
  }

  Future<void> _handleNext() async {
    debugPrint('UseTab handleNext called - vocabIndex: ${widget.state.vocabIndex}');

    if (!_isLastVocabPage) {
      await _speakNextVocab();
      if (widget.state.nextStepId == TeachWordConstants.steps['goToUse${widget.state.vocabIndex + 2}']) {
        widget.controllers.navigateToNextTab();
      }
      ref.read(teachWordStateProvider.notifier).updateVocabIndex(
        widget.state.vocabIndex + 1
      );
    }
  }

  void _handleContinue() {
    debugPrint('UseTab handleContinue called');
    setState(() {
      _userMessage.value = '';
      _showContinueButton.value = false;
    });
  }

  Future<void> _speakPreviousVocab() async {
    String prevVocab = widget.wordsPhrase['vocab${widget.state.vocabIndex}'];
    String prevSentence = widget.wordsPhrase['sentence${widget.state.vocabIndex}'];
    final tts = ref.read(ttsProvider);
    await tts.speak("$prevVocab。$prevSentence");
  }

  Future<void> _speakNextVocab() async {
    String nextVocab = widget.wordsPhrase['vocab${widget.state.vocabIndex + 2}'];
    String nextSentence = widget.wordsPhrase['sentence${widget.state.vocabIndex + 2}'];
    final tts = ref.read(ttsProvider);
    await tts.speak("$nextVocab。$nextSentence");
  }
}

class VocabContentController extends ChangeNotifier {
  final VoidCallback onVocabComplete;
  final VoidCallback onVocabRetry;
  
  bool _isAnswerCorrect = false;
  String _selectedWord = '';
  String _message = '';

  bool get isAnswerCorrect => _isAnswerCorrect;
  String get selectedWord => _selectedWord;
  String get message => _message;

  VocabContentController({
    required this.onVocabComplete,
    required this.onVocabRetry,
  });

  void setAnswer(bool isCorrect, String word) {
    debugPrint('Setting answer: $word is correct? $isCorrect');
    _isAnswerCorrect = isCorrect;
    _selectedWord = word;
    _message = isCorrect ? '答對了！' : '再試試！';
    notifyListeners();

    if (isCorrect) {
      onVocabComplete();
    } else {
      onVocabRetry();
    }
  }

  void reset() {
    debugPrint('Resetting vocab content controller');
    _isAnswerCorrect = false;
    _selectedWord = '';
    _message = '';
    notifyListeners();
  }
}

class AudioController {
  final VoidCallback onPlayComplete;
  final Function(String) onPlayError;
  
  final FlutterTts _tts = FlutterTts();
  bool _isPlaying = false;

  AudioController({
    required this.onPlayComplete,
    required this.onPlayError,
  }) {
    _initializeTts();
  }

  void _initializeTts() {
    _tts.setLanguage("zh-tw");
    _tts.setSpeechRate(0.5);
    _tts.setVolume(1.0);
    _tts.setCompletionHandler(() {
      debugPrint('Audio playback complete');
      _isPlaying = false;
      onPlayComplete();
    });
    _tts.setErrorHandler((msg) {
      debugPrint('Audio playback error: $msg');
      _isPlaying = false;
      onPlayError(msg.toString());
    });
  }

  Future<void> speak(String text) async {
    debugPrint('TTS speaking: $text');
    if (_isPlaying) {
      await _tts.stop();
      debugPrint('Stopped current TTS playback');
    }
    _isPlaying = true;
    await _tts.speak(text);
  }

  Future<void> stop() async {
    debugPrint('Stopping audio playback');
    if (_isPlaying) {
      await _tts.stop();
      _isPlaying = false;
    }
  }

  void dispose() {
    debugPrint('Disposing TTS controller');
    _tts.stop();
  }
}

class _UseTabContent extends StatelessWidget {
  final TeachWordState state;

  final VocabContentController contentController;
  final AudioController audioController;

    required this.screenInfo,

          ),
        ),

          right: screenInfo.fontSize,
          bottom: screenInfo.fontSize,
          child: _buildPageIndicator(),
        ),
      ],
    );

  Widget _buildContent() {
    return state.isBpmf
        ? BopomofoVocabContent(
            word: state.word,
            vocab: _getCurrentVocab(),
            sentence: _getCurrentSentence(),
            audioController: audioController,
            onComplete: contentController.onVocabComplete, // Link the vocab completion here
          )
        : WordVocabContent(
            vocab: _getCurrentVocab(),
            meaning: _getCurrentMeaning(),
            sentence: _getCurrentSentence(),
            vocab2: _getNextVocab(),
            contentController: contentController,
            audioController: audioController,
            onComplete: contentController.onVocabComplete, // Link completion callback here
          );
  }

    return Text(
      "${state.vocabIndex + 1} / ${state.vocabCnt}",
      style: TextStyle(
        fontSize: screenInfo.fontSize * 0.75,
        color: Colors.black,
      ),
    );

  String _getCurrentMeaning() => wordsPhrase['meaning${state.vocabIndex + 1}'] ?? '';
  String _getCurrentSentence() => wordsPhrase['sentence${state.vocabIndex + 1}'] ?? '';

// State persistence manager
class UseTabStateManager {
  static const String _keyPrefix = 'use_tab_state_';
  final String _wordId;
  
  UseTabStateManager(this._wordId);

  // Keys for different state aspects
  String get _vocabIndexKey => '${_keyPrefix}vocab_index_$_wordId';
  String get _completedVocabsKey => '${_keyPrefix}completed_vocabs_$_wordId';
  String get _currentProgressKey => '${_keyPrefix}current_progress_$_wordId';

  // Save state
  Future<void> saveState({
    required int vocabIndex,
    required List<String> completedVocabs,
    required Map<String, dynamic> currentProgress,
  }) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setInt(_vocabIndexKey, vocabIndex);
    await prefs.setStringList(_completedVocabsKey, completedVocabs);
    await prefs.setString(_currentProgressKey, json.encode(currentProgress));
  }

  // Restore state
  Future<UseTabRestoredState?> restoreState() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      
      if (!prefs.containsKey(_vocabIndexKey)) {
        return null;
      }

      return UseTabRestoredState(
        vocabIndex: prefs.getInt(_vocabIndexKey) ?? 0,
        completedVocabs: prefs.getStringList(_completedVocabsKey) ?? [],
        currentProgress: json.decode(prefs.getString(_currentProgressKey) ?? '{}'),
      );
    } catch (e) {
      debugPrint('Error restoring state: $e');
      return null;
    }
  }

  // Clear state
  Future<void> clearState() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove(_vocabIndexKey);
    await prefs.remove(_completedVocabsKey);
    await prefs.remove(_currentProgressKey);
  }
}

class UseTabState extends State<UseTab> with WidgetsBindingObserver {
  late final UseTabStateManager _stateManager;
  late final TabFocusHandler _focusHandler;
  late final AudioStateManager _audioManager;
  late final ErrorHandler _errorHandler;
  Timer? _autoSaveTimer;
  bool _isRestoringState = false;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _initializeManagers();
    _setupAutoSave();
    _restoreState();
  }

  void _initializeManagers() {
    _stateManager = UseTabStateManager(widget.state.word);
    _focusHandler = TabFocusHandler(
      onFocusGained: _handleTabFocused,
      onFocusLost: _handleTabUnfocused,
    );
    _audioManager = AudioStateManager(
      onInterruption: _handleAudioInterruption,
      onError: _handleAudioError,
    );
    _errorHandler = ErrorHandler(
      onError: _handleError,
      onRecovery: _handleErrorRecovery,
    );
  }

  void _setupAutoSave() {
    _autoSaveTimer = Timer.periodic(
      const Duration(seconds: 30),
      (_) => _saveCurrentState(),
    );
  }

  Future<void> _restoreState() async {
    setState(() => _isRestoringState = true);
    
    try {
      final restoredState = await _stateManager.restoreState();
      if (restoredState != null) {
        _applyRestoredState(restoredState);
      }
    } catch (e) {
      _errorHandler.handleError(
        ErrorType.stateRestoration,
        error: e,
        context: context,
      );
    } finally {
      setState(() => _isRestoringState = false);
    }
  }

  void _applyRestoredState(UseTabRestoredState state) {
    setState(() {
      widget.state.vocabIndex = state.vocabIndex;
      // Apply other restored state properties
    });
  }

  Future<void> _saveCurrentState() async {
    if (_isRestoringState) return;

    try {
      await _stateManager.saveState(
        vocabIndex: widget.state.vocabIndex,
        completedVocabs: _getCompletedVocabs(),
        currentProgress: _getCurrentProgress(),
      );
    } catch (e) {
      _errorHandler.handleError(
        ErrorType.stateSaving,
        error: e,
        context: context,
      );
    }
  }

  // Edge case handling
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    switch (state) {
      case AppLifecycleState.paused:
        _handleAppPaused();
        break;
      case AppLifecycleState.resumed:
        _handleAppResumed();
        break;
      default:
        break;
    }
  }

  void _handleAppPaused() {
    _audioManager.pauseAudio();
    _saveCurrentState();
  }

  void _handleAppResumed() {
    _restoreState();
  }

  void _handleTabFocused() {
    _audioManager.resumeAudio();
    _restoreState();
  }

  void _handleTabUnfocused() {
    _audioManager.pauseAudio();
    _saveCurrentState();
  }

  void _handleAudioInterruption() {
    _audioManager.handleInterruption();
  }

  void _handleAudioError(AudioError error) {
    _errorHandler.handleError(
      ErrorType.audio,
      error: error,
      context: context,
    );
  }

  @override
  Widget build(BuildContext context) {
    if (_isRestoringState) {
      return _buildLoadingState();
    }

    return ErrorBoundary(
      onError: _handleError,
      child: TabFocusWrapper(
        handler: _focusHandler,
        child: _UseTabContent(
          state: widget.state,
          wordsPhrase: widget.wordsPhrase,
          audioManager: _audioManager,
          errorHandler: _errorHandler,
          onStateChange: _handleStateChange,
        ),
      ),
    );
  }

  Widget _buildLoadingState() {
    return const Center(
      child: CircularProgressIndicator(),
    );
  }

  void _handleStateChange() {
    _saveCurrentState();
  }

  @override
  void dispose() {
    _autoSaveTimer?.cancel();
    _audioManager.dispose();
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }
}

// Error handling
class ErrorHandler {
  final Function(ErrorType, {dynamic error, BuildContext? context}) onError;
  final Function()? onRecovery;

  ErrorHandler({
    required this.onError,
    this.onRecovery,
  });

  void handleError(ErrorType type, {dynamic error, BuildContext? context}) {
    debugPrint('Error of type $type: $error');
    onError(type, error: error, context: context);
  }

  Future<void> attemptRecovery() async {
    try {
      await onRecovery?.call();
    } catch (e) {
      handleError(ErrorType.recovery, error: e);
    }
  }
}

// Audio state management
class AudioStateManager {
  final VoidCallback onInterruption;
  final Function(AudioError) onError;
  
  bool _isPlaying = false;
  final FlutterTts _tts = FlutterTts();
  AudioFocus? _audioFocus;

  AudioStateManager({
    required this.onInterruption,
    required this.onError,
  }) {
    _initializeAudio();
  }

  Future<void> _initializeAudio() async {
    try {
      await _tts.setLanguage("zh-tw");
      await _tts.setSpeechRate(0.5);
      await _tts.setVolume(1.0);
      
      _tts.setCompletionHandler(() {
        _isPlaying = false;
      });

      _tts.setErrorHandler((msg) {
        _isPlaying = false;
        onError(AudioError(msg.toString()));
      });

      _initializeAudioFocus();
    } catch (e) {
      onError(AudioError(e.toString()));
    }
  }

  Future<void> _initializeAudioFocus() async {
    _audioFocus = await AudioFocus.requestFocus(
      onFocusChange: (focus) {
        if (focus == AudioFocusState.loss) {
          pauseAudio();
          onInterruption();
        }
      },
    );
  }

  Future<void> playAudio(String text) async {
    if (!_isPlaying) {
      try {
        final focusResult = await _audioFocus?.requestFocus();
        if (focusResult == AudioFocusRequest.granted) {
          _isPlaying = true;
          await _tts.speak(text);
        }
      } catch (e) {
        onError(AudioError(e.toString()));
      }
    }
  }

  Future<void> pauseAudio() async {
    if (_isPlaying) {
      try {
        await _tts.stop();
        _isPlaying = false;
        await _audioFocus?.abandonFocus();
      } catch (e) {
        onError(AudioError(e.toString()));
      }
    }
  }

  void dispose() {
    _tts.stop();
    _audioFocus?.abandonFocus();
  }
}

// Tab focus management
class TabFocusHandler {
  final VoidCallback onFocusGained;
  final VoidCallback onFocusLost;
  bool _hasFocus = false;

  TabFocusHandler({
    required this.onFocusGained,
    required this.onFocusLost,
  });

  void handleFocusChange(bool focused) {
    if (focused != _hasFocus) {
      _hasFocus = focused;
      if (focused) {
        onFocusGained();
      } else {
        onFocusLost();
      }
    }
  }
}

class TeachWordConstants {
  static const int tabCount = 4;
  static const double strokeWidth = 2.0;
  static const Duration animationDuration = Duration(milliseconds: 300);    
  
  // Tab indices for direct reference
  static const int lookTabIndex = 0;
  static const int listenTabIndex = 1;
  static const int writeTabIndex = 2;
  static const int useTabIndex = 3;

  // Step progression map
  static const Map<String, int> steps = {
    'goToListen': 0,
    'goToWrite': 1,
    'seeAnimation': 2,
    'practiceWithBorder1': 3,
    'practiceWithBorder2': 4,
    'practiceWithBorder3': 5,
    'turnBorderOff': 6,
    'practiceWithoutBorder1': 7,
    'goToUse1': 8,
    'goToUse2': 9,
  };

  // Helper method to get tab index based on step
  static int getTabIndexForStep(int stepId) {
    if (stepId < steps['goToListen']!) {
      return lookTabIndex;
    } else if (stepId < steps['goToWrite']!) {
      return listenTabIndex;
    } else if (stepId < steps['goToUse1']!) {
      return writeTabIndex;
    } else {
      return useTabIndex;
    }
  }
}

@immutable
class TeachWordState {
  final String word;
  final bool wordExists;
  final bool svgFileExists;
  final bool isLearned;
  final int nextStepId;
  final int practiceTimeLeft;
  final Map<String, dynamic> wordData;
  final int vocabCount;
  final int vocabIndex;
  final bool isBpmf;
  final bool img1Exists;
  final bool img2Exists;  
  final int currentTabIndex; 

  const TeachWordState({
    required this.word,
    this.wordExists = false,
    this.svgFileExists = false,
    this.isLearned = false,
    this.nextStepId = 0,
    this.practiceTimeLeft = 4,
    required this.wordData,
    this.vocabCount = 0,
    this.vocabIndex = 0,
    this.isBpmf = false,
    this.img1Exists = false, 
    this.img2Exists = false,     
    this.currentTabIndex = 0,
  });

  // Write tab control helpers
  bool get canShowStrokeOrder => 
    svgFileExists && (isLearned || nextStepId == TeachWordConstants.steps['seeAnimation']);

  bool get canShowWriteButton =>
    svgFileExists && (isLearned || 
    (nextStepId > TeachWordConstants.steps['seeAnimation']! &&
     nextStepId <= TeachWordConstants.steps['turnBorderOff']!));

  bool get canToggleOutline =>
    svgFileExists && (isLearned || nextStepId == TeachWordConstants.steps['turnBorderOff']);

  bool get canNavigateNext =>
    isLearned || nextStepId < TeachWordConstants.steps['goToUse2']!;

  bool get canNavigatePrevious => true;  // Always allow going back

  TeachWordState copyWith({
    String? word,
    bool? wordExists,
    bool? svgFileExists,
    bool? isLearned,
    int? nextStepId,
    int? practiceTimeLeft,
    Map<String, dynamic>? wordData,
    int? vocabCount,
    int? vocabIndex,
    bool? isBpmf,
    bool? img1Exists,
    bool? img2Exists,    
    int? currentTabIndex,
  }) {
    return TeachWordState(
      word: word ?? this.word,
      wordExists: wordExists ?? this.wordExists,
      svgFileExists: svgFileExists ?? this.svgFileExists,
      isLearned: isLearned ?? this.isLearned,
      nextStepId: nextStepId ?? this.nextStepId,
      practiceTimeLeft: practiceTimeLeft ?? this.practiceTimeLeft,
      wordData: wordData ?? this.wordData,
      vocabCount: vocabCount ?? this.vocabCount,
      vocabIndex: vocabIndex ?? this.vocabIndex,
      isBpmf: isBpmf ?? this.isBpmf,
      img1Exists: img1Exists ?? this.img1Exists,
      img2Exists: img2Exists ?? this.img2Exists,  
      currentTabIndex: currentTabIndex ?? this.currentTabIndex,    
    );
  }

  @override
  String toString() {
    return 'TeachWordState(word: $word, exists: $wordExists, svg: $svgFileExists, '
           'learned: $isLearned, stepId: $nextStepId, practiceLeft: $practiceTimeLeft)';
  }
}

      isLast: (wordIndex == wordsStatus.length - 1),
      onLeftClicked: _buildLeftClickHandler(context, ref),
      onRightClicked: _buildRightClickHandler(context, ref),
    );
  }

  Widget _buildWordCard() {
    return WordCard(
      unitId: unitId,
      unitTitle: unitTitle,
      wordsStatus: wordsStatus,
      wordsPhrase: wordsPhrase,
      wordIndex: wordIndex,
      sizedBoxWidth: screenInfo.fontSize * 10,
      sizedBoxHeight: screenInfo.fontSize * 4.0,
      fontSize: screenInfo.fontSize * 1.2,
      isBpmf: state.isBpmf,
      isVertical: false,
      disable: true,
    );
  }

  VoidCallback? _buildLeftClickHandler(BuildContext context, WidgetRef ref) {
    if (wordIndex == 0) return null;

    return () => navigateWithProvider(
      context,
      '/teachWord',
      ref,
      arguments: {
        'unitId': unitId,
        'unitTitle': unitTitle,
        'wordsStatus': wordsStatus,
        'wordsPhrase': wordsPhrase,
        'wordIndex': wordIndex - 1,
      },
    );
  }

  VoidCallback? _buildRightClickHandler(BuildContext context, WidgetRef ref) {
    if (wordIndex == wordsStatus.length - 1) return null;

    return () => navigateWithProvider(
      context,
      '/teachWord',
      ref,
      arguments: {
        'unitId': unitId,
        'unitTitle': unitTitle,
        'wordsStatus': wordsStatus,
        'wordsPhrase': wordsPhrase,
        'wordIndex': wordIndex + 1,
      },
    );
  }

  Widget _buildSentence(double fontSize) {

class TeachWordView extends ConsumerStatefulWidget {
  final int unitId;
  final String unitTitle;
  final List<WordStatus> wordsStatus;
  final List<Map<String, dynamic>> wordsPhrase;
  final int wordIndex;

  const TeachWordView({
    super.key,
    required this.unitId,
    required this.unitTitle,
    required this.wordsStatus,
    required this.wordsPhrase,
    required this.wordIndex,
  });

  @override
  TeachWordViewState createState() => TeachWordViewState();
}

class TeachWordViewState extends ConsumerState<TeachWordView> with TickerProviderStateMixin {
  late final TeachWordControllers controllers;
  bool svgFileExist = false;
  String word = '';
  bool isLearned = false;
  // bool _firstNullStroke = true;
  bool _showErrorDialog = false;
  int _numberOfNullStroke = 0;
  final int _maxNumberOfNullStroke = 30;

  @override
  void initState() {
    super.initState();
    word = widget.wordsStatus[widget.wordIndex].word;
    isLearned = widget.wordsStatus[widget.wordIndex].learned;
    debugPrint('TeachWordView initState for word: $word, learned: $isLearned');
    
    // Initialize controllers with the current TickerProvider
    controllers = TeachWordControllers(
      ref,
      TabController(
        length: TeachWordConstants.tabCount,
        vsync: this,
      ),
    );
    
    _initializeView();
  }

  // Q：wordData 是否應該變成這個 class 的 global variable？
  Future<void> _initializeView() async {
    // Cast the Map to the correct type
    final wordData = Map<String, dynamic>.from(widget.wordsPhrase[widget.wordIndex]);

    // Initialize the state first. Wrap state initialization in a post-frame callback
    WidgetsBinding.instance.addPostFrameCallback((_) {
      // Initialize the state first
      ref.read(teachWordStateProvider.notifier).initializeWord(
        word,
        wordData,
        isLearned,
      );
    });

    // Then initialize stroke order
    await _initializeStrokeOrder();
  }

  // Q：state 是否應該變成這個 class 的 global variable？
  Future<void> _initializeStrokeOrder() async {
    final state = ref.read(teachWordStateProvider); // Q: When was wordExists set?
    if (!state.wordExists) return;

    try {
      final json = await readJson(state.word);
      if (json.isNotEmpty) {
        await controllers.initializeStrokeController(
          json,
          this, // Pass the ticker provider
          _handleQuizCompletion,
        );
        ref.read(teachWordStateProvider.notifier).setSvgFileExists(true);
      }
    } catch (e) {
      debugPrint('_initializeStrokeOrder error: $e');
      WidgetsBinding.instance.addPostFrameCallback((_) {
        ref.read(teachWordStateProvider.notifier).setSvgFileExists(false);
      });
    }
  }

  void _handleQuizCompletion(QuizSummary summary) {
    final notifier = ref.read(teachWordStateProvider.notifier);
    final state = ref.read(teachWordStateProvider);
    final screenInfo = ref.read(screenInfoProvider);  // Get current screenInfo

    if (state.nextStepId >= TeachWordConstants.steps['practiceWithBorder1']! &&
        state.nextStepId <= TeachWordConstants.steps['turnBorderOff']!) {
      notifier.decrementPracticeTimeLeft();
      notifier.incrementNextStepId();
      
      controllers.showPracticeCompletionToast(
        state.practiceTimeLeft,
        state.nextStepId == TeachWordConstants.steps['turnBorderOff'],
        screenInfo.fontSize,  // Pass current fontSize
      );
    } else if (state.nextStepId == TeachWordConstants.steps['practiceWithoutBorder1']) {
      notifier.decrementPracticeTimeLeft();
      notifier.incrementNextStepId();
      notifier.markWordAsLearned(); // Note: 未來要改成用一用結束後，才標記為已學習。
      _updateWordStatusInDatabase();
      controllers.showCompletionToast(screenInfo.fontSize);  // Pass current fontSize
    }
  }
  
  Future<void> _updateWordStatusInDatabase() async {
    final status = widget.wordsStatus[widget.wordIndex];
    status.learned = true;
    await WordStatusProvider().updateWordStatus(status: status);
  }

  Future<String> readJson(String word) async {
    // Define the noSvgList. This list needs to be updated every new school semester
    List<String> noSvgList = [
      '吔', '姍', '媼', '嬤', '履', '搧', '枴', '椏', '欓', '汙',
      '溼', '漥', '痠', '礫', '粄', '粿', '綰', '蓆', '襬', '譟',
      '踖', '踧', '鎚', '鏗', '鏘', '陳', '颺', '齒'
    ];

    String tempWord = word;

    // If the word is in the noSvgList, set svgFileExist to false and use '學' as a fallback
    if (noSvgList.contains(word)) {
      tempWord = '學';
      svgFileExist = false;
    } else {
      svgFileExist = true;
    }

    debugPrint('readJson() word: $tempWord, svgFileExist: $svgFileExist');

    // Proceed with loading the SVG file. If svgFileExist is false, 
    // load “學” so that 看一看、聽一聽、用一用 still work.  
    try {
      final String response = await rootBundle.loadString('lib/assets/svg/$tempWord.json');
      return response.replaceAll("\"", "'");
    } catch (e) {
      svgFileExist = false;
      debugPrint('readJson() error: $e, svgFileExist: $svgFileExist');
      return ''; // Return an empty string in case of error
    }
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(teachWordStateProvider);
    final screenInfo = ref.watch(screenInfoProvider);
    
    if (state.word == '' || screenInfo.fontSize == 0) {
      debugPrint('TeachWordView build() word: $word, fontSize: ${screenInfo.fontSize}');
      return Container();
    } 
    
    if (controllers.strokeOrderAnimationController == null) {
      debugPrint('TeachWordView build() strokeOrderAnimationController is null');
      return _buildErrorWidget(state.word, screenInfo.fontSize);
    }

    return DefaultTabController(
      length: TeachWordConstants.tabCount,
      child: Scaffold(
        appBar: TeachWordAppBar(
          screenInfo: screenInfo,
          controllers: controllers,
          state: state,
          unitId: widget.unitId,
          unitTitle: widget.unitTitle,
        ),
        body: TabBarView(
          physics: const NeverScrollableScrollPhysics(),
          controller: controllers.tabController,
          children: [
            LookTab(
              controllers: controllers,
              screenInfo: screenInfo,
              state: state,
            ),
            ListenTab(
              controllers: controllers,
              screenInfo: screenInfo,
              state: state,
            ),
            WriteTab(
              controllers: controllers,
              screenInfo: screenInfo,
              state: state,
            ),            
            UseTab(
              controllers: controllers,
              screenInfo: screenInfo,
              state: state,
              wordsPhrase: Map<String, dynamic>.from(widget.wordsPhrase[widget.wordIndex]),
            ),
          ],
        ),
        bottomNavigationBar: TeachWordBottomBar(
          screenInfo: screenInfo,
          controllers: controllers,
          state: state,
          unitId: widget.unitId,
          unitTitle: widget.unitTitle,
          wordsStatus: widget.wordsStatus,
          wordsPhrase: widget.wordsPhrase,
          wordIndex: widget.wordIndex,
        ),
      ),
    );
  }

  Widget _buildErrorWidget(String word, double fontSize) {
    if (_numberOfNullStroke < _maxNumberOfNullStroke) {
      _numberOfNullStroke++;
      debugPrint('buildErrorWidget() word: $word, numberOfNullStroke: $_numberOfNullStroke');      
      return Container();
    } else {
      _showErrorDialog = true;
      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (_showErrorDialog) {
          controllers.showErrorDialog(
            context,
            '','「$word」沒有筆順2。請截圖回報。謝謝！',
            fontSize,
          );
          _numberOfNullStroke = 0;
          _showErrorDialog = false;
        }
      });
      return Container();
    }
  }

  @override
  Ticker createTicker(TickerCallback onTick) => Ticker(onTick);
}
